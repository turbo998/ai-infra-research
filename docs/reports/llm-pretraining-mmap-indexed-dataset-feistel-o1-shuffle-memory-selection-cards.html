# Maps any logical index i in [0, N) to a unique shuffled index in [0, N) in O(1) RAM:
def feistel_permute(i: int, n: int, seed: int, mask: int, half_bits: int) -> int:
    while True:
        l, r = i >> half_bits, i & ((1 << half_bits) - 1)
        for rnd in range(4):
            l, r = r, l ^ (hash64(r, seed + rnd) & ((1 << half_bits) - 1))
        i = (l << half_bits) | r
        if i < n: return i
