// ⛔ Anti-Pattern: Raw @torch.inference_mode() output fed to nn.Linear
@torch.inference_mode()

    return self.vision_tower(pixel_values) # is_inference() == True!

vit_embeds = self.encode_images(images)
proj = self.mm_projector(vit_embeds)
# ^^^ CRASHES in nn.Linear: ctx.save_for_backward(vit_embeds)
# RuntimeError: Inference tensors cannot be saved for backward!
↓ Single Exit-Boundary .clone() Bridge ↓
// ✅ Production Pattern: inference_mode() inside + .clone() outside!
def encode_images(self, pixel_values):
    with torch.inference_mode():
        vit_embeds = self.vision_tower(pixel_values) # 960 ops fast!
    # Outside inference_mode(): .clone() allocates a normal TensorImpl
    # with a fresh VariableVersion counter (0.02 ms for 16 MB):
    return vit_embeds.clone()

proj = self.mm_projector(self.encode_images(images)) # 100% Safe!

7 / 7 · MY TAKE

Architect Verdict: Use inference_mode() for Serving/Frozen Towers & .clone() at Training Boundaries

✅ Good fit
• Pure LLM serving (`vLLM / SGLang` workers use `@torch.inference_mode()`)
• Frozen VLM `ViT` / Audio encoders with a single exit `.clone()` before the trainable projector
• RL rollout generation in `verl / OpenRLHF`
⛔ Not for
• Using `torch.no_grad()` in pure inference servers
• Passing uncloned `is_inference() == True` tensors into trainable modules or in-place mutations
