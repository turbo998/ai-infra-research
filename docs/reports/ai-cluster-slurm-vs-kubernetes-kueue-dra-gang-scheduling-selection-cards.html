# 1. Kubernetes 1.32+ DRA (Dynamic Resource Allocation): Atomic GPU + RDMA NIC Same-PCIe-Switch Claim
apiVersion: resource.k8s.io/v1beta1
kind: ResourceClaimTemplate
metadata:
  name: aligned-gpu-rdma-claim
spec:
  spec:
    devices:
      requests:
        - name: gpu
          deviceClassName: gpu.nvidia.com
          count: 4
        - name: rdma-nic
          deviceClassName: rdma.mellanox.com
          count: 4
      constraints:
        - requests: ["gpu", "rdma-nic"]
          matchAttribute: "pcie.io/switch-id" # Guarantees 100% GPUDirect RDMA PCIe Switch alignment!
