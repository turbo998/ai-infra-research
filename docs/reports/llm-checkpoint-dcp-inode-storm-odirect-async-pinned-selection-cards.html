Persistent Pinned AsyncStager + Two-Phase Rename

import torch.distributed.checkpoint as dcp from torch.distributed.checkpoint.staging import DefaultStager, StagerConfig ↓ # 1. Reuse persistent Pinned Host staging memory across steps (48ms D2H!): stager = DefaultStager(StagerConfig(use_pinned_memory=True, use_shared_memory=True)) # 2. Write to .tmp_step_N with O_DIRECT / FADV_DONTNEED, then atomic rename: ckpt_future = dcp.async_save( state_dict, checkpoint_id=f"/mnt/3fs/ckpt/.tmp_step_{step}", async_stager=stager, process_group=dedup_shard_pg )
