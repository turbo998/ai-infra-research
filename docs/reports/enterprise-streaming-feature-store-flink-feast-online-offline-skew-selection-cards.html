Tiled Hopping Pane Partial Aggregates & Temporal ASOF Point-in-Time Join

-- Temporal ASOF Point-in-Time Join (Zero Future Data Leakage)
SELECT l.tx_id, l.user_id, l.label_fraud, f.failed_tx_5m, f.spend_zscore_30d
FROM chargeback_labels l
ASOF JOIN feature_archive_iceberg f
  ON l.user_id = f.user_id
  AND f.event_timestamp <= l.tx_timestamp
  AND f.event_timestamp >= l.tx_timestamp - INTERVAL '48 HOURS';
