# 1. Prefill 阶段 (MInference): 用末尾 64 个 Query 探测垂直锚点列 (Vertical) 与斜线 (Slash)
def minference_vertical_slash_prefill(Q, K, V, v_topk: int = 64, s_topk: int = 96):
    # 仅计算最后 64 个 Query 与全量 128K Key 的近似注意力图 (~2.5ms)
    approx_attn = torch.softmax((Q[:, :, -64:, :] @ K.transpose(-1, -2)) / sqrt_d, dim=-1)
    vertical_indices = torch.topk(approx_attn.sum(dim=-2), k=v_topk, dim=-1).indices
    slash_indices = detect_top_diagonals(approx_attn, k=s_topk)
    # 仅对 ~12% 活跃块启动 Triton 稀疏 FlashAttention，生成完整 128K KV (14.80s -> 1.82s!)
    return triton_vertical_slash_flash_attn(Q, K, V, vertical_indices, slash_indices)
↓ Decode 阶段 (Quest): 页面包络框 [K_min, K_max] 非破坏性打分挑选 Top-256 页 (<answer>) ↓
# 2. Decode 阶段 (Quest): 保留全量 8,000 页 (16 tok/页)，每步动态挑选 Top-256 页 (4,096 tok)
def quest_page_envelope_decode_step(q_t, kv_pages, k_min_env, k_max_env, top_pages: int = 256):
    # q_t: [B, H, 1, D], k_min_env / k_max_env: [B, H, 8000, D]
    # 通道级精确内积上界: 若 q_j > 0 取 q_j * K_max_j，若 q_j < 0 取 q_j * K_min_j (耗时 0.35ms)
    upper_bound_scores = torch.maximum(q_t * k_max_env, q_t * k_min_env).sum(dim=-1)
    top256_page_ids = torch.topk(upper_bound_scores, k=top_pages, dim=-1).indices
    # 仅将这 256 个活跃页 (4,096 tokens = 3.1% 流量) Gather 进 Tensor Core 精算 Attention！
    return flashinfer_paged_decode_subset(q_t, kv_pages, page_table=top256_page_ids)

✅ Good Fit (强烈推荐场景)

  • • 128K~256K 多轮代码库问答 / 法律合同 / 财报深度分析 → Tier 4 (MInference + Quest)：TTFT 压至 1.82s，TPOT 压至 9.8ms，5 轮追问召回 98.9%。
  • • 长轨迹 Agent 多步推理 → Tier 4 + Host DRAM 分层：包络索引常驻 HBM，冷 KV 页放 DRAM 按需换入。

⛔ Not For (严禁踩坑场景)

  • • 严禁在多轮交互问答中开启 H2O / SnapKV 永久性 KV 删除 (Tier 3)：用户第 2 轮一旦换个章节提问就会失忆。
  • • 严禁对代码、SQL 或财务报表做 LLMLingua 硬删词 (Tier 2)：会破坏关键语法符号与 Prefix Cache 命中。
