class ZeroBubbleLinear(torch.autograd.Function):
    @staticmethod
    def backward(ctx, grad_output):
        x, weight = ctx.saved_tensors
        grad_input = torch.matmul(grad_output, weight)  # B (Critical Path!)
        WeightGradStore.put(x, grad_output, weight.main_grad)  # Defer W!
        return grad_input, None
