# 1. Deterministic SQLGlot AST Gate (Read-Only + Mandatory Partition Key) import sqlglot from sqlglot import exp def validate_generated_sql(sql: str, dialect: str = "snowflake") -> str: tree = sqlglot.parse_one(sql, read=dialect) assert isinstance(tree, exp.Select), "BLOCKED: Only SELECT statements allowed!" assert "dt" in sql.lower() or "order_date" in sql.lower(), "Missing partition filter!" return tree.limit(1000).sql(dialect=dialect)

# 2. Abort Runaway Table Scans Before Execution & Enforce User OAuth OBO RLS plan = warehouse.explain_dry_run(validated_sql, user_obo_token=jwt_token) if plan.estimated_bytes_scanned > 10 * 1024**3: # > 10 GB cap raise ScanCircuitBreakerError("Query scans >10GB; narrow date range!") return warehouse.execute(validated_sql, user_obo_token=jwt_token, timeout_sec=15)
