# 1. 入队层：全局 Radix 前缀字典序 + 长度同质双维分桶器 (Two-Level Offline Binner)
def build_bucketed_micro_shards(prompts: list[dict], shard_size: int = 512) -> list[list[dict]]:
    # 第一维：提取前 512 词或 system_template_id 的前缀哈希，确保相同模板 100% 聚在一起
    # 第二维：在同一模板内部按 (input_len + max_output_tokens) 升序排列，消除长短混批空洞
    sorted_prompts = sorted(
        prompts,
        key=lambda r: (r["prefix_hash_bucket"], r["input_tokens"] + r["max_output_tokens"])
    )
    return [sorted_prompts[i : i + shard_size] for i in range(0, len(sorted_prompts), shard_size)]
