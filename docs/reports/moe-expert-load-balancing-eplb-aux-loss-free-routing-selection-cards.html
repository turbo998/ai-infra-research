# 1. 训练期 (Tier 3): Aux-Loss-Free 免辅助损失动态偏置路由 (零干扰梯度!)
def aux_loss_free_moe_forward(x, gate_weight, expert_bias_b, gamma: float = 1e-3):
    s = torch.sigmoid(x @ gate_weight)              # 原始 Token-to-Expert 亲和度分数
    # 关键：选路时加上无梯度偏置 b_i，但乘回专家输出时严格使用原始未偏移的 s！
    _, topk_idx = torch.topk(s.detach() + expert_bias_b, k=8, dim=-1)
    gating_weights = torch.gather(s, dim=-1, index=topk_idx)  # 梯度仅流经真实亲和度 s
    with torch.no_grad():
        load_i = torch.bincount(topk_idx.view(-1), minlength=256).float()
        expert_bias_b += gamma * torch.sign(load_i.mean() - load_i)  # 忙则降 b_i，闲则升 b_i

# 2. 推理期 (Tier 4): EPLB 注水法分配 32 个冗余副本 + 后台 P2P 零停机切换
def eplb_plan_replicas(ema_expert_loads: list[float], num_logical: int = 256, extra_slots: int = 32):
    replicas = [1] * num_logical
    for _ in range(extra_slots):
        # 每次挑当前 (单副本平均负载 = load_i / replicas_i) 最高的热门专家增加 1 个物理副本
